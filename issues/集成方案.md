📊 指标中心 (Metrics Hub) 全任务分析

  一、任务概览与依赖关系

  任务依赖树:
  T1 (OTLP接收器) ────┬─→ T2 (归一化器) ─→ T3 (双写存储) ──┬─→ T4 (读取API) ──┐
                      │                                      │                 │
                      └──────────────────────────────────────┴─→ T5 (聚合/保留) │
                                                             │                 │
  T6 (自动部署Agent) ────────────────────────────────────────┘                 │
                                                                               │
  T7 (安全加固) ←─────────────────────────────────────────────────────────────┤
  T8 (性能优化) ←─────────────────────────────────────────────────────────────┘

  二、各任务详细分析

  T1 - OTLP接收器 (基础设施)

  - 端口: gRPC(4317) + HTTP(4318) + 管理(8080)
  - 核心组件: OtlpGrpcServer, OtlpHttpController, IngestQueue
  - 关键技术: Spring Boot, gRPC, 内存队列, Micrometer
  - 输出: 快速入队,暴露健康检查

  T2 - 归一化器 (数据处理)

  - 输入: OTLP Export Payload
  - 输出: List<MetricSample> 内部模型
  - 功能:
    - 单位归一化 (bytes/seconds)
    - 标签白名单 (server.id/region/group/env)
    - 支持指标: cpu/mem/disk/filesystem/network/load
  - 核心组件: OtlpDecoder, Normalizer

  T3 - 双写存储 (持久化层)

  - Redis: 热数据 + 最新值
    - Key: metrics:latest:{serverId} (HASH)
    - 可选 ZSET 时间窗口
  - TSDB: PostgreSQL/Timescale 原始时序数据
    - 表: server_metric_samples
  - 核心组件: WritePipeline, RedisWriter, TsdbWriter
  - 数据库迁移: Flyway V1__create_server_metric_samples.sql

  T4 - 读取API (查询服务)

  - 接口:
    - GET /monitoring/server/{id}/metrics?from&to&step&fields
    - GET /monitoring/servers/metrics?fields=cpu,mem,net
  - 智能路由: 近实时读Redis, 历史读TSDB
  - 性能目标: 近实时≤300ms, 大盘≤1.5s
  - 输出: OpenAPI 文档

  T5 - 聚合与保留 (数据生命周期)

  - 聚合粒度: 15s原始 → 5m → 1h
  - 保留策略: 30-90天可配
  - 查询优化: 自动选择聚合粒度
  - 核心组件: RollupJob (CRON定时)
  - 数据库迁移: V2__rollup_views.sql

  T6 - Agent自动部署 (运维自动化)

  - 触发: ServerCreatedEvent
  - 流程: 预检 → 上传 → 配置渲染 → systemd启动
  - 组件:
    - otelcol (OpenTelemetry Collector)
    - bootstrap.sh 引导脚本
    - otelcol.yaml.tmpl 配置模板
  - 进度推送: WebSocket /topic/server-status
  - 重试策略: 1m/5m/15m

  T7 - 安全加固 (生产就绪)

  - 传输安全: TLS/mTLS (gRPC + HTTP)
  - 认证: Bearer Token + JWKS验证
  - 加密: 敏感信息at-rest加密
  - 日志脱敏: 无明文敏感信息
  - 限流: 按Agent做限流

  T8 - 性能优化 (规模化)

  - 目标规模: 1000台主机 × 10-15s周期
  - 优化点:
    - 批处理与背压控制
    - Redis/DB连接池
    - 线程池调优
  - 工具: otlp_load_gen.sh, LoadSimulator
  - 监控: P95时延/QPS/CPU/GC

  A. MonitoringController 改造

  // src/main/java/com/cmict/internalpaas/controller/MonitoringController.java

  @RestController
  @RequestMapping("/monitoring")
  public class MonitoringController {

      @Autowired
      private MetricsHubClient metricsHubClient; // 新增

      @Autowired
      private MonitoringService legacyMonitoringService; // 保留

      @GetMapping("/server/{id}/metrics")
      public ResponseEntity<?> getServerMetrics(
              @PathVariable Long id,
              @RequestParam(required = false) Long from,
              @RequestParam(required = false) Long to) {

          // 策略1: 优先使用Metrics Hub
          if (metricsHubEnabled && from != null) {
              try {
                  return metricsHubClient.getMetrics(id, from, to);
              } catch (Exception e) {
                  log.warn("Metrics Hub unavailable, fallback to SSH", e);
              }
          }

          // 策略2: 降级到SSH轮询
          return legacyMonitoringService.getServerMetrics(id);
      }
  }

  // MonitoringController智能路由升级
  @GetMapping("/server/{id}/metrics")
  public ResponseEntity<?> getMetrics(
          @PathVariable Long id,
          @RequestParam Long from,
          @RequestParam Long to,
          @RequestParam(required = false) String step) {

      Duration timeRange = Duration.between(
          Instant.ofEpochMilli(from),
          Instant.ofEpochMilli(to)
      );

      // 智能选择数据源
      if (timeRange.toMinutes() <= 5) {
          // 最近5分钟: Redis热数据
          return metricsHubClient.getRealtimeMetrics(id, from, to);
      } else if (timeRange.toDays() <= 1) {
          // 1天内: 原始数据
          return metricsHubClient.getRawMetrics(id, from, to, "15s");
      } else if (timeRange.toDays() <= 7) {
          // 7天内: 5分钟聚合
          return metricsHubClient.getRollupMetrics(id, from, to, "5m");
      } else {
          // 7天以上: 1小时聚合
          return metricsHubClient.getRollupMetrics(id, from, to, "1h");
      }
  }

  // metrics-hub-client/src/main/java/com/cmict/metrics/client/MetricsHubClient.java
  package com.cmict.metrics.client;

  import org.springframework.beans.factory.annotation.Value;
  import org.springframework.boot.web.client.RestTemplateBuilder;
  import org.springframework.stereotype.Component;
  import org.springframework.web.client.RestTemplate;

  import java.time.Duration;
  import java.util.Map;

  @Component
  public class MetricsHubClient {

      private final RestTemplate restTemplate;
      private final String baseUrl;

      public MetricsHubClient(
              RestTemplateBuilder builder,
              @Value("${metrics.hub.base-url}") String baseUrl) {
          this.baseUrl = baseUrl;
          this.restTemplate = builder
                  .setConnectTimeout(Duration.ofSeconds(5))
                  .setReadTimeout(Duration.ofSeconds(10))
                  .build();
      }

      /**
       * 获取服务器实时指标 (最近5分钟, 走Redis)
       */
      public MetricsResponse getRealtimeMetrics(Long serverId, Long from, Long to) {
          String url = String.format(
              "%s/api/v1/metrics/server/%d/realtime?from=%d&to=%d",
              baseUrl, serverId, from, to
          );
          return restTemplate.getForObject(url, MetricsResponse.class);
      }

      /**
       * 获取历史原始数据 (走TSDB)
       */
      public MetricsResponse getRawMetrics(
              Long serverId, Long from, Long to, String step) {
          String url = String.format(
              "%s/api/v1/metrics/server/%d/raw?from=%d&to=%d&step=%s",
              baseUrl, serverId, from, to, step
          );
          return restTemplate.getForObject(url, MetricsResponse.class);
      }

      /**
       * 获取聚合数据 (走聚合表)
       */
      public MetricsResponse getRollupMetrics(
              Long serverId, Long from, Long to, String rollup) {
          String url = String.format(
              "%s/api/v1/metrics/server/%d/rollup?from=%d&to=%d&rollup=%s",
              baseUrl, serverId, from, to, rollup
          );
          return restTemplate.getForObject(url, MetricsResponse.class);
      }

      /**
       * 批量获取多服务器最新指标
       */
      public Map<Long, MetricsSnapshot> getLatestMetrics(List<Long> serverIds) {
          String url = baseUrl + "/api/v1/metrics/servers/latest";
          return restTemplate.postForObject(
              url,
              Map.of("serverIds", serverIds),
              new ParameterizedTypeReference<Map<Long, MetricsSnapshot>>() {}
          );
      }

      /**
       * 健康检查
       */
      public boolean isHealthy() {
          try {
              String url = baseUrl + "/actuator/health";
              var response = restTemplate.getForObject(url, Map.class);
              return "UP".equals(response.get("status"));
          } catch (Exception e) {
              return false;
          }
      }
  }

  6.2 改造现有MonitoringController

  // src/main/java/com/cmict/internalpaas/controller/MonitoringController.java
  package com.cmict.internalpaas.controller;

  import com.cmict.internalpaas.service.MonitoringService;
  import com.cmict.metrics.client.MetricsHubClient;
  import lombok.extern.slf4j.Slf4j;
  import org.springframework.beans.factory.annotation.Autowired;
  import org.springframework.beans.factory.annotation.Value;
  import org.springframework.http.ResponseEntity;
  import org.springframework.web.bind.annotation.*;

  import java.time.Duration;
  import java.time.Instant;

  @Slf4j
  @RestController
  @RequestMapping("/monitoring")
  public class MonitoringController {

      @Autowired
      private MetricsHubClient metricsHubClient;

      @Autowired
      private MonitoringService legacyMonitoringService;

      @Value("${metrics.hub.enabled:false}")
      private boolean metricsHubEnabled;

      @Value("${metrics.hub.fallback-to-ssh:true}")
      private boolean fallbackEnabled;

      /**
       * 获取单服务器指标 (兼容现有API)
       * 前端调用保持不变: GET /monitoring/server/{id}/metrics
       */
      @GetMapping("/server/{id}/metrics")
      public ResponseEntity<?> getServerMetrics(
              @PathVariable Long id,
              @RequestParam(required = false) Long from,
              @RequestParam(required = false) Long to,
              @RequestParam(required = false) String fields) {

          // 如果未启用Metrics Hub, 直接用SSH
          if (!metricsHubEnabled) {
              log.debug("Metrics Hub disabled, using SSH polling");
              return ResponseEntity.ok(legacyMonitoringService.getServerMetrics(id));
          }

          // 如果没有时间范围参数, 返回最新值
          if (from == null || to == null) {
              return getLatestMetrics(id);
          }

          try {
              // 根据时间范围智能路由
              return ResponseEntity.ok(routeMetricsQuery(id, from, to));
          } catch (Exception e) {
              log.error("Metrics Hub query failed for server {}", id, e);

              // 降级到SSH轮询
              if (fallbackEnabled) {
                  log.warn("Falling back to SSH polling for server {}", id);
                  return ResponseEntity.ok(legacyMonitoringService.getServerMetrics(id));
              }

              throw e;
          }
      }

      /**
       * 获取多服务器最新指标 (用于仪表板)
       * 前端调用: GET /monitoring/servers/metrics?fields=cpu,mem,net
       */
      @GetMapping("/servers/metrics")
      public ResponseEntity<?> getServersMetrics(
              @RequestParam(required = false) String fields) {

          if (!metricsHubEnabled) {
              return ResponseEntity.ok(legacyMonitoringService.getAllServersMetrics());
          }

          try {
              // 获取所有服务器ID
              List<Long> serverIds = serverRepository.findAll()
                      .stream()
                      .map(Server::getId)
                      .collect(Collectors.toList());

              // 批量查询最新指标
              Map<Long, MetricsSnapshot> metrics =
                      metricsHubClient.getLatestMetrics(serverIds);

              return ResponseEntity.ok(metrics);
          } catch (Exception e) {
              log.error("Batch metrics query failed", e);

              if (fallbackEnabled) {
                  return ResponseEntity.ok(legacyMonitoringService.getAllServersMetrics());
              }

              throw e;
          }
      }

      /**
       * 健康检查 (用于前端检测Metrics Hub状态)
       */
      @GetMapping("/health")
      public ResponseEntity<?> getHealth() {
          boolean hubHealthy = metricsHubEnabled && metricsHubClient.isHealthy();

          return ResponseEntity.ok(Map.of(
              "metricsHub", hubHealthy ? "UP" : "DOWN",
              "sshPolling", "UP",
              "fallbackEnabled", fallbackEnabled
          ));
      }

      /**
       * 智能路由: 根据时间范围选择数据源
       */
      private Object routeMetricsQuery(Long serverId, Long from, Long to) {
          Duration timeRange = Duration.between(
              Instant.ofEpochMilli(from),
              Instant.ofEpochMilli(to)
          );

          log.debug("Query time range: {} for server {}", timeRange, serverId);

          // 最近5分钟: Redis热数据
          if (timeRange.toMinutes() <= 5) {
              log.debug("Using Redis (hot data)");
              return metricsHubClient.getRealtimeMetrics(serverId, from, to);
          }

          // 1天内: 原始数据 (15s粒度)
          else if (timeRange.toDays() <= 1) {
              log.debug("Using TSDB raw data (15s step)");
              return metricsHubClient.getRawMetrics(serverId, from, to, "15s");
          }

          // 7天内: 5分钟聚合
          else if (timeRange.toDays() <= 7) {
              log.debug("Using 5m rollup");
              return metricsHubClient.getRollupMetrics(serverId, from, to, "5m");
          }

          // 7天以上: 1小时聚合
          else {
              log.debug("Using 1h rollup");
              return metricsHubClient.getRollupMetrics(serverId, from, to, "1h");
          }
      }

      /**
       * 获取最新值 (无时间范围参数时)
       */
      private Object getLatestMetrics(Long serverId) {
          // 尝试从Redis获取最新值
          try {
              return metricsHubClient.getLatestMetrics(List.of(serverId))
                      .get(serverId);
          } catch (Exception e) {
              log.warn("Failed to get latest metrics from Hub", e);

              if (fallbackEnabled) {
                  return legacyMonitoringService.getServerMetrics(serverId);
              }

              throw e;
          }
      }
  }

  6.3 Agent自动部署集成

  // src/main/java/com/cmict/internalpaas/service/AgentDeployService.java
  package com.cmict.internalpaas.service;

  import com.cmict.internalpaas.model.Server;
  import com.cmict.internalpaas.repository.ServerRepository;
  import lombok.extern.slf4j.Slf4j;
  import org.springframework.beans.factory.annotation.Autowired;
  import org.springframework.beans.factory.annotation.Value;
  import org.springframework.messaging.simp.SimpMessagingTemplate;
  import org.springframework.scheduling.annotation.Async;
  import org.springframework.stereotype.Service;

  import java.io.IOException;
  import java.nio.file.Files;
  import java.nio.file.Path;
  import java.util.Map;

  @Slf4j
  @Service
  public class AgentDeployService {

      @Autowired
      private ServerRepository serverRepository;

      @Autowired
      private SshConnectionService sshService;

      @Autowired
      private SimpMessagingTemplate messagingTemplate;

      @Value("${metrics.hub.otlp-endpoint}")
      private String otlpEndpoint;

      @Value("${metrics.agent.bootstrap-script}")
      private String bootstrapScriptPath;

      @Value("${metrics.agent.otelcol-config}")
      private String otelcolConfigPath;

      /**
       * 异步部署Agent
       */
      @Async
      public void deploy(Long serverId) {
          Server server = serverRepository.findById(serverId)
                  .orElseThrow(() -> new RuntimeException("Server not found: " + serverId));

          try {
              log.info("Starting agent deployment for server {}", serverId);
              sendProgress(serverId, "STARTED", "Starting agent deployment...");

              // 1. 预检
              sendProgress(serverId, "CHECKING", "Checking SSH connectivity...");
              preCheck(server);

              // 2. 上传文件
              sendProgress(serverId, "UPLOADING", "Uploading agent files...");
              uploadAgentFiles(server);

              // 3. 渲染配置
              sendProgress(serverId, "CONFIGURING", "Rendering configuration...");
              String config = renderConfig(server);
              sshService.uploadContent(server, "/tmp/otelcol.yaml", config);

              // 4. 执行安装脚本
              sendProgress(serverId, "INSTALLING", "Installing otelcol...");
              String installCmd = String.format(
                  "SERVER_ID=%d OTLP_ENDPOINT=%s sh /tmp/bootstrap.sh otelcol",
                  serverId, otlpEndpoint
              );
              sshService.executeCommand(server, installCmd);

              // 5. 验证
              sendProgress(serverId, "VERIFYING", "Verifying agent status...");
              String statusCmd = "systemctl status otelcol --no-pager";
              String status = sshService.executeCommand(server, statusCmd);

              if (status.contains("active (running)")) {
                  sendProgress(serverId, "SUCCESS", "Agent deployed successfully");
                  log.info("Agent deployed successfully on server {}", serverId);
              } else {
                  throw new RuntimeException("Agent not running after deployment");
              }

          } catch (Exception e) {
              log.error("Agent deployment failed for server {}", serverId, e);
              sendProgress(serverId, "FAILED", "Deployment failed: " + e.getMessage());

              // 可选: 触发重试
              scheduleRetry(serverId);
          }
      }

      /**
       * 预检: SSH连接、权限、磁盘空间
       */
      private void preCheck(Server server) throws Exception {
          // 检查SSH连接
          if (!sshService.testConnection(server)) {
              throw new RuntimeException("SSH connection failed");
          }

          // 检查磁盘空间 (至少需要500MB)
          String dfCmd = "df -m /opt | tail -1 | awk '{print $4}'";
          String available = sshService.executeCommand(server, dfCmd).trim();
          if (Integer.parseInt(available) < 500) {
              throw new RuntimeException("Insufficient disk space");
          }

          // 检查sudo权限
          String sudoCmd = "sudo -n true 2>&1";
          String result = sshService.executeCommand(server, sudoCmd);
          if (result.contains("password")) {
              throw new RuntimeException("Sudo password required");
          }
      }

      /**
       * 上传Agent文件
       */
      private void uploadAgentFiles(Server server) throws Exception {
          // 上传bootstrap脚本
          Path bootstrapScript = Path.of(bootstrapScriptPath);
          String scriptContent = Files.readString(bootstrapScript);
          sshService.uploadContent(server, "/tmp/bootstrap.sh", scriptContent);

          // 赋执行权限
          sshService.executeCommand(server, "chmod +x /tmp/bootstrap.sh");

          // 上传otelcol二进制 (如果需要)
          // sshService.uploadFile(server, "agent/otelcol", "/tmp/otelcol");
      }

      /**
       * 渲染配置模板
       */
      private String renderConfig(Server server) throws IOException {
          String template = Files.readString(Path.of(otelcolConfigPath));

          // 替换模板变量
          return template
                  .replace("${SERVER_ID}", server.getId().toString())
                  .replace("${SERVER_NAME}", server.getName())
                  .replace("${OTLP_ENDPOINT}", otlpEndpoint)
                  .replace("${REGION}", server.getRegion() != null ? server.getRegion() : "default")
                  .replace("${ENV}", server.getEnvironment() != null ? server.getEnvironment() : "dev");
      }

      /**
       * 发送进度到前端
       */
      private void sendProgress(Long serverId, String status, String message) {
          Map<String, Object> progress = Map.of(
              "type", "AGENT_DEPLOY",
              "serverId", serverId,
              "status", status,
              "message", message,
              "timestamp", System.currentTimeMillis()
          );

          messagingTemplate.convertAndSend("/topic/server-status", progress);
      }

      /**
       * 失败重试调度
       */
      private void scheduleRetry(Long serverId) {
          // TODO: 实现重试逻辑 (1m/5m/15m)
          log.warn("Retry not implemented yet for server {}", serverId);
      }
  }

  sudo systemctl enable ${AGENT_NAME}
  sudo systemctl start ${AGENT_NAME}

  sleep 3
  if sudo systemctl is-active --quiet ${AGENT_NAME}; then
      echo "✓ ${AGENT_NAME} installed and running"
      exit 0

              this.updateDataSourceIndicator(health);
          } catch (error) {
              console.warn('Metrics Hub health check failed', error);
              this.metricsHubEnabled = false;
          }
      }

      /**
       * 加载服务器指标 (自动路由到最优数据源)
       */
      async loadServerMetrics(serverId, timeRange) {
          const { from, to } = this.calculateTimeRange(timeRange);

          try {
              // 调用统一API, 后端自动路由
              const response = await fetch(
                  `/monitoring/server/${serverId}/metrics?from=${from}&to=${to}`
              );

              if (!response.ok) {
                  throw new Error(`HTTP ${response.status}`);
              }

              const data = await response.json();

              // 显示数据源提示
              this.showDataSourceHint(data.source, timeRange);

              return data;

          } catch (error) {
              console.error('Failed to load metrics', error);
              this.showErrorNotification('数据加载失败，请稍后重试');
              throw error;
          }
      }

      /**
       * 实时监控 (WebSocket订阅)
       */
      subscribeRealtimeMetrics(serverId) {
          if (!this.stompClient) {
              this.connectWebSocket();
          }

          // 订阅服务器状态更新
          this.stompClient.subscribe(`/topic/server-status`, (message) => {
              const update = JSON.parse(message.body);

              if (update.serverId === serverId) {
                  this.handleRealtimeUpdate(update);
              }
          });

          // 如果启用Metrics Hub, 也可以通过轮询Redis获取最新值
          if (this.metricsHubEnabled) {
              this.startRealtimePolling(serverId);
          }
      }

      /**
       * 轮询最新指标 (用于近实时场景)
       */
      startRealtimePolling(serverId) {
          if (this.pollingInterval) {
              clearInterval(this.pollingInterval);
          }

          this.pollingInterval = setInterval(async () => {
              try {
                  const response = await fetch(
                      `/monitoring/server/${serverId}/metrics`
                  );
                  const data = await response.json();

                  // 更新图表
                  this.updateRealtimeChart(data);

              } catch (error) {
                  console.error('Realtime polling error', error);
              }
          }, 15000); // 每15秒更新一次
      }

      /**
       * 显示数据源提示
       */
      showDataSourceHint(source, timeRange) {
          const hints = {
              'redis': `🟢 实时数据 (Redis) - 最近${timeRange}` ,
              'tsdb-raw': `🔵 原始数据 (TSDB) - 15秒粒度`,
              'tsdb-5m': `🟡 聚合数据 (5分钟) - 优化查询性能` ,

          };

          const hint = hints[source] || '数据源未知';

          if (indicator) {

          }
      }

      /**
       * 计算时间范围
       */
      calculateTimeRange(preset) {
          const now = Date.now();
          const ranges = {
              '5m': 5 * 60 * 1000,
              '1h': 60 * 60 * 1000,
              '6h': 6 * 60 * 60 * 1000,
              '1d': 24 * 60 * 60 * 1000,
              '7d': 7 * 24 * 60 * 60 * 1000,
              '30d': 30 * 24 * 60 * 60 * 1000
          };

          const duration = ranges[preset] || ranges['1h'];

          return {
              from: now - duration,
              to: now
          };
      }

  package com.cmict.internalpaas.service;

  import com.cmict.internalpaas.model.Server;
  import com.cmict.internalpaas.repository.ServerRepository;
  import lombok.extern.slf4j.Slf4j;
  import org.springframework.beans.factory.annotation.Autowired;
  import org.springframework.stereotype.Service;

  import java.time.LocalDateTime;
  import java.util.HashMap;
  import java.util.Map;

  @Slf4j
  @Service
  public class AgentService {

      @Autowired
      private ServerRepository serverRepository;

      @Autowired
      private SshConnectionService sshService;

      @Autowired
      private AgentDeployService deployService;

      /**
       * 获取Agent状态
       */
      public Map<String, Object> getAgentStatus(Long serverId) {
          Server server = serverRepository.findById(serverId)
                  .orElseThrow(() -> new RuntimeException("Server not found"));

          Map<String, Object> status = new HashMap<>();

          try {
              // 通过SSH检查Agent进程
              String checkCmd = "systemctl is-active otelcol 2>/dev/null || echo 'not-installed'";
              String result = sshService.executeCommand(server, checkCmd).trim();

              if ("active".equals(result)) {
                  status.put("status", "RUNNING");

                  // 获取版本
                  String versionCmd = "/opt/metrics-agent/otelcol --version 2>/dev/null | head -1";
                  String version = sshService.executeCommand(server, versionCmd).trim();
                  status.put("version", version);

                  // 获取配置
                  String configCmd = "grep 'collection_interval' /opt/metrics-agent/config.yaml | awk '{print $2}'";
                  String interval = sshService.executeCommand(server, configCmd).trim();
                  status.put("collectionInterval", interval.replace("s", ""));

              } else if ("not-installed".equals(result)) {
                  status.put("status", "NOT_INSTALLED");
              } else {
                  status.put("status", "STOPPED");
              }

              // 从数据库获取部署信息
              status.put("deployedAt", server.getAgentDeployedAt());
              status.put("lastReportTime", server.getLastMetricReportTime());
              status.put("otlpEndpoint", server.getOtlpEndpoint());

          } catch (Exception e) {
              log.error("Failed to get agent status for server {}", serverId, e);
              status.put("status", "ERROR");
              status.put("error", e.getMessage());
          }

          return status;
      }
