```java
/**
 * 获取服务器最新监控数据 (仅使用 Hub)
 */
@GetMapping("/server/{id}/metrics")
@ResponseBody
public ResponseEntity<?> getServerMetrics(@PathVariable Long id) {
    try {
        // 检查服务器是否存在
        if (!serverService.findById(id).isPresent()) {
            return ResponseEntity.status(404).body(Map.of(
                "error", "服务器不存在",
                "serverId", id
            ));
        }

        // 检查 Hub 是否可用
        if (metricsHubClient == null || !metricsHubClient.isAvailable()) {
            logger.error("❌ Metrics Hub 不可用 - serverId: {}", id);
            return ResponseEntity.status(503).body(Map.of(
                "error", "监控服务暂时不可用，请稍后重试",
                "hint", "请确保 OTLP Agent 已部署并正常运行",
                "serverId", id
            ));
        }

        // 从 Hub 读取数据
        logger.debug("从 Metrics Hub 读取数据 - serverId: {}", id);
        ServerMetrics metrics = metricsHubClient.getLatestMetrics(id);

        if (metrics == null) {
            logger.warn("Hub 返回空数据 - serverId: {}", id);
            return ResponseEntity.status(404).body(Map.of(
                "error", "暂无监控数据",
                "hint", "请确保 OTLP Agent 正在运行并上报数据",
                "serverId", id
            ));
        }

        logger.info("✅ 数据来源: Metrics Hub - serverId: {}", id);

        return ResponseEntity.ok(Map.of(
            "dataSource", "hub",
            "metrics", metrics,
            "timestamp", System.currentTimeMillis()
        ));

    } catch (Exception e) {
        logger.error("获取监控数据异常 - serverId: {}, error: {}", id, e.getMessage(), e);
        return ResponseEntity.internalServerError().body(Map.of(
            "error", e.getMessage(),
            "serverId", id
        ));
    }
}

/**
 * 获取所有服务器指标 (仅使用 Hub)
 */
@GetMapping("/servers/metrics")
@ResponseBody
public ResponseEntity<Map<String, Object>> getAllServersMetrics() {
    try {
        List<Server> servers = serverService.getActiveServers();
        Map<String, ServerMetrics> metricsMap = new HashMap<>();
        int hubCount = 0;
        int errorCount = 0;

        // 检查 Hub 是否可用
        boolean hubAvailable = metricsHubClient != null && metricsHubClient.isAvailable();

        if (!hubAvailable) {
            logger.warn("⚠️ Metrics Hub 不可用，无法获取服务器指标");
            return ResponseEntity.status(503).body(Map.of(
                "error", "监控服务暂时不可用",
                "hint", "请检查 Metrics Hub 服务状态"
            ));
        }

        // 从 Hub 获取所有服务器数据
        for (Server server : servers) {
            try {
                ServerMetrics metrics = metricsHubClient.getLatestMetrics(server.getId());
                if (metrics != null) {
                    metricsMap.put(String.valueOf(server.getId()), metrics);
                    hubCount++;
                } else {
                    errorCount++;
                    logger.debug("服务器 {} 暂无 Hub 数据", server.getId());
                }
            } catch (Exception e) {
                errorCount++;
                logger.error("获取服务器 {} Hub数据失败: {}", server.getId(), e.getMessage());
            }
        }

        logger.info("批量获取指标完成 - Hub: {}, 错误: {}, 总数: {}", 
            hubCount, errorCount, servers.size());

        Map<String, Object> response = new HashMap<>();
        response.put("metrics", metricsMap);
        response.put("dataSource", "hub");
        response.put("timestamp", System.currentTimeMillis());
        response.put("statistics", Map.of(
            "total", servers.size(),
            "hubCount", hubCount,
            "errorCount", errorCount
        ));

        return ResponseEntity.ok(response);

    } catch (Exception e) {
        logger.error("批量获取监控数据异常: {}", e.getMessage(), e);
        return ResponseEntity.internalServerError().body(Map.of(
            "error", e.getMessage()
        ));
    }
}
```

## 七、Step 6: 历史数据迁移 (可选)

### 7.1 目标

将 H2 数据库中的历史 `ServerMetrics` 数据迁移到 Hub 的 PostgreSQL/TimescaleDB

### 7.2 迁移脚本

```java
package com.cmict.internalpaas.migration;

import org.springframework.stereotype.Component;
import org.springframework.boot.CommandLineRunner;

/**
 * H2 历史数据迁移工具
 * 
 * 用法:
 * 1. 临时恢复 H2 数据源配置
 * 2. 运行此脚本
 * 3. 验证数据迁移
 * 4. 移除 H2 依赖
 */
@Component
public class H2ToHubDataMigration implements CommandLineRunner {

    @Autowired
    private ServerMetricsRepository h2Repository; // H2 仓库
    
    @Autowired
    private JdbcTemplate hubJdbcTemplate; // Hub PostgreSQL
    
    @Override
    public void run(String... args) throws Exception {
        if (!Boolean.getBoolean("migration.enabled")) {
            logger.info("数据迁移未启用,跳过");
            return;
        }
        
        logger.info("开始迁移 H2 历史数据到 Hub TSDB...");
        
        // 1. 读取 H2 所有数据
        List<ServerMetrics> allMetrics = h2Repository.findAll();
        logger.info("从 H2 读取 {} 条记录", allMetrics.size());
        
        // 2. 按服务器分组
        Map<Long, List<ServerMetrics>> byServer = allMetrics.stream()
            .collect(Collectors.groupingBy(ServerMetrics::getServerId));
        
        // 3. 批量插入 Hub TSDB
        int successCount = 0;
        int errorCount = 0;
        
        for (Map.Entry<Long, List<ServerMetrics>> entry : byServer.entrySet()) {
            Long serverId = entry.getKey();
            List<ServerMetrics> metrics = entry.getValue();
            
            try {
                batchInsertToHub(serverId, metrics);
                successCount += metrics.size();
                logger.info("服务器 {} 迁移完成: {} 条记录", serverId, metrics.size());
            } catch (Exception e) {
                errorCount += metrics.size();
                logger.error("服务器 {} 迁移失败: {}", serverId, e.getMessage(), e);
            }
        }
        
        logger.info("数据迁移完成 - 成功: {}, 失败: {}", successCount, errorCount);
    }
    
    private void batchInsertToHub(Long serverId, List<ServerMetrics> metrics) {
        String sql = """
            INSERT INTO server_metric_samples (
                server_id, metric_name, metric_value, unit,
                timestamp_ms, labels
            ) VALUES (?, ?, ?, ?, ?, ?::jsonb)
            ON CONFLICT (server_id, metric_name, timestamp_ms) DO NOTHING
            """;
        
        hubJdbcTemplate.batchUpdate(sql, new BatchPreparedStatementSetter() {
            @Override
            public void setValues(PreparedStatement ps, int i) throws SQLException {
                ServerMetrics m = metrics.get(i);
                ps.setLong(1, serverId);
                
                // 拆分为多个指标
                insertMetric(ps, "system.cpu.usage", m.getCpuUsage(), "percent");
                insertMetric(ps, "system.memory.usage", m.getMemoryUsage(), "percent");
                insertMetric(ps, "system.disk.usage", m.getDiskUsage(), "percent");
                // ... 其他指标
            }
            
            @Override
            public int getBatchSize() {
                return metrics.size();
            }
        });
    }
}
```
